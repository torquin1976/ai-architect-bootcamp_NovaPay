# 1. Create a working folder and drop server.js into it
mkdir novapay-monolith && cd novapay-monolith
# (copy the server.js shown above into this folder)

# 2. Start Postgres + Redis in Docker
docker run -d --name np-pg -e POSTGRES_USER=np -e POSTGRES_PASSWORD=np \
  -e POSTGRES_DB=novapay -p 5432:5432 postgres:15
docker run -d --name np-redis -p 6379:6379 redis:7

# 3. Create the txns table (wait ~5s for Postgres to be ready first)
docker exec -i np-pg psql -U np -d novapay -c \
  "CREATE TABLE IF NOT EXISTS txns(
     id TEXT PRIMARY KEY, merchant TEXT, amount NUMERIC,
     status TEXT, created_at TIMESTAMPTZ DEFAULT now());"

# 4. Install deps and run
npm init -y
npm install express body-parser pg ioredis
node server.js
# → NovaPay monolith :3000

# 5. Smoke test — should return {token, status:"AUTHORIZED", amount:4999}
curl -X POST localhost:3000/auth \
  -H 'content-type: application/json' \
  -d '{"card":"4111111111111111","amount":4999,"merchantId":"m_42","idempotencyKey":"k1"}'

curl localhost:3000/health
# → {"status":"ok","pid":...,"uptime":...}
